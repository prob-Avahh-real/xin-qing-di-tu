```javascript
    { name: "起风了", artist: "买辣椒也用券", url: "..." },
    { name: "平凡之路", artist: "朴树", url: "..." }
  ]
};

// 随机选择
function getRandomMusic(emotion) {
  const list = MUSIC_PLAYLIST[emotion];
  return list[Math.floor(Math.random() * list.length)];
}
```

```javascript
// 根据时间段推荐
```

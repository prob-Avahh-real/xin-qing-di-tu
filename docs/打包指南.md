### 方法 2: 自动打包脚本

**创建打包脚本** `package.ps1`:

```powershell
# 心情地图 - 自动打包脚本
# 使用方法：.\package.ps1

$timestamp = Get-Date -Format "yyyyMMdd"
$packageName = "心情地图 - 提交版_$timestamp"
$packageDir = Join-Path $PSScriptRoot $packageName

# 创建目录
if (Test-Path $packageDir) {
    Remove-Item $packageDir -Recurse -Force
}
New-Item -ItemType Directory -Path $packageDir | Out-Null

Write-Host "📦 开始打包：$packageName" -ForegroundColor Green

# 核心文件列表
$coreFiles = @(
    "心情地图 - 导航中心.html",
    "business-pro-rich-routes.html",
    "心情地图 - 统一导航.html",
    "门票预订 - 完整版.html",
    "points-mall-app.html",
    "wishing-pool.html",
    "knowledge-base.html",
    "poi-detail.html",
    "游玩指引 - 用户留存系统.html",
    "merchant-partnership.html",
    "admin-dashboard.html",
    "小红书网红店搜索.html",
    "demo-judge-fixed.html",
    "功能快速测试.html",
    "manifest.json",
    "sw.js"
)

# 文档文件
$docFiles = @(
    "README.md",
    "提交前检查清单.md",
    "推广策略.md",
    "青年用户体验地图.md",
    "PWA-APP 安装指南.md"
)

# 复制文件
Write-Host "📋 复制核心文件..." -ForegroundColor Yellow
foreach ($file in $coreFiles) {
    if (Test-Path $file) {
        Copy-Item $file -Destination $packageDir
        Write-Host "  ✅ $file" -ForegroundColor Green
    } else {
        Write-Host "  ⚠️  $file (不存在)" -ForegroundColor Yellow
    }
}

Write-Host "📋 复制文档文件..." -ForegroundColor Yellow
foreach ($file in $docFiles) {
    if (Test-Path $file) {
        Copy-Item $file -Destination $packageDir
        Write-Host "  ✅ $file" -ForegroundColor Green
    } else {
        Write-Host "  ⚠️  $file (不存在)" -ForegroundColor Yellow
    }
}

# 复制 icons 目录
if (Test-Path "icons") {
    Copy-Item "icons" -Destination $packageDir -Recurse
    Write-Host "  ✅ icons/" -ForegroundColor Green
}

# 压缩
Write-Host "📦 创建 ZIP 压缩包..." -ForegroundColor Yellow
$zipPath = "$packageDir.zip"
Compress-Archive -Path "$packageDir/*" -DestinationPath $zipPath -Force

Write-Host "✅ 打包完成！" -ForegroundColor Green
Write-Host "📦 文件位置：$zipPath" -ForegroundColor Cyan

# 显示统计
$fileCount = (Get-ChildItem $packageDir -File).Count
$totalSize = (Get-ChildItem $packageDir -File | Measure-Object Length -Sum).Sum / 1KB
Write-Host "📊 文件数量：$fileCount" -ForegroundColor Cyan
Write-Host "📊 总大小：[Math]::Round($totalSize, 2) KB" -ForegroundColor Cyan
```

**使用方法**:
```powershell
cd E:\hackathon-travel-experience-main
.\package.ps1
```
